import socketio
import numpy as np
import sounddevice as sd
import time
import threading

SERVER_URL = "http://DESKTOP-FTC.local:5001"
sio = socketio.Client(reconnection=True, reconnection_attempts=0, reconnection_delay=1)

playing_state = False
start_time = 0
last_audio_time = 0

def silence_monitor():
    global playing_state, start_time, last_audio_time
    while True:
        if playing_state and (time.time() - last_audio_time > 0.8):
            duration = round(time.time() - start_time, 1)
            try:
                sio.emit('pi_status', {'status': 'finished', 'duration': duration})
            except: pass
            playing_state = False
            start_time = 0
        time.sleep(0.1)

@sio.on('play_on_pi')
def on_audio(data):
    global playing_state, start_time, last_audio_time
    last_audio_time = time.time()
    if not playing_state:
        playing_state = True
        start_time = time.time()
        sio.emit('pi_status', {'status': 'playing'})
    try:
        audio_array = np.frombuffer(data, dtype=np.int16)
        sd.play(audio_array, 44100)
    except: pass

@sio.event
def connect():
    print("Connected!")
    sio.emit('pi_status', {'status': 'connected'})

if __name__ == '__main__':
    threading.Thread(target=silence_monitor, daemon=True).start()
    while True:
        try:
            if not sio.connected:
                sio.connect(SERVER_URL)
            sio.wait()
        except:
            time.sleep(2)
